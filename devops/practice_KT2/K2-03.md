# K2-03. Compose и сохранность данных

docker compose -f compose.student.yaml config
services:
  api:
    image: devops-student:v1
    environment:
      DATA_DIR: /data
      ROLE: api
    volumes:
      - source: requests
        target: /data
        type: volume
  web:
    depends_on:
      api:
        condition: service_healthy
    environment:
      API_URL: http://api:8000
      ROLE: web
    image: devops-student:v1
    ports:
      - host_ip: 127.0.0.1
        published: "8080"
        target: 8000

docker compose -f compose.student.yaml up -d
[+] up 2/2
 ✔ Container devops-course-main-api-1 Healthy                                                                                                    4.7s
 ✔ Container devops-course-main-web-1 Started                                                                                                    4.9s

docker compose -f compose.student.yaml exec web python -c "import urllib.request, json; data = json.dumps({'kind': 'screen', 'key': 'test1'}).encode(); req = urllib.request.Request('http://api:8000/api/requests', data=data, headers={'Content-Type': 'application/json'}, method='POST'); print(urllib.request.urlopen(req).read().decode())"
{"request": {"id": 1, "kind": "screen", "created": "2026-10-17T08:10:37.355853+00:00"}, "instance": "237496ce3c55"}

docker compose -f compose.student.yaml up -d --no-deps --force-recreate api
[+] up 1/1
 ✔ Container devops-course-main-api-1 Started                                                                                                   10.7s

docker compose -f compose.student.yaml exec web python -c "import urllib.request; print(urllib.request.urlopen('http://api:8000/api/requests').read().decode())"
{"requests": [{"id": 1, "kind": "screen", "created": "2026-10-17T08:10:37.355853+00:00"}], "total": 1, "instance": "da6b44b6fff8"}

docker compose -f compose.student.yaml exec web python -c "import urllib.request, json; data = json.dumps({'kind': 'battery', 'key': 'test2'}).encode(); req = urllib.request.Request('http://api:8000/api/requests', data=data, headers={'Content-Type': 'application/json'}, method='POST'); print(urllib.request.urlopen(req).read().decode())"
{"request": {"id": 2, "kind": "battery", "created": "2026-10-17T08:11:19.049720+00:00"}, "instance": "da6b44b6fff8"}

Почему нельзя хранить базу только в изменяемом слое контейнера?
При пересоздании контейнера (--force-recreate) его файловая система удаляется. Данные сохраняются только в томе requests.
